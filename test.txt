#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <time.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

#define N_FFT     128
#define N_SYM     62
#define N_SAMPLES 19200
#define RUNS      100
#define FILE_NAME "signal.pcm"

void gen_pss(double pss[3][N_SYM]) {
    int u[] = {25, 29, 34};
    for (int nid2 = 0; nid2 < 3; nid2++)
        for (int n = 0; n < N_SYM; n++)
            pss[nid2][n] = -M_PI * u[nid2] * (n + 31) * (n + 32) / 63;
}

int detect_pss(const double* signal, int len, double pss[3][N_SYM],
               int* pss_pos, double corr_out[3]) {
    double best_corr = 0;
    int best_nid2 = 0;
    int best_pos = 0;

    for (int pos = 0; pos < len - N_SYM * 2; pos++) {
        for (int nid2 = 0; nid2 < 3; nid2++) {
            double cr = 0, ci = 0;
            for (int n = 0; n < N_SYM; n++) {
                double ph = pss[nid2][n];
                cr += signal[pos + 2*n]     * cos(ph) - signal[pos + 2*n + 1] * sin(ph);
                ci += signal[pos + 2*n]     * sin(ph) + signal[pos + 2*n + 1] * cos(ph);
            }
            double c = sqrt(cr*cr + ci*ci);
            if (c > best_corr) {
                best_corr = c;
                best_nid2 = nid2;
                best_pos = pos;
            }
        }
    }

    for (int nid2 = 0; nid2 < 3; nid2++) {
        double cr = 0, ci = 0;
        for (int n = 0; n < N_SYM; n++) {
            double ph = pss[nid2][n];
            cr += signal[best_pos + 2*n]     * cos(ph) - signal[best_pos + 2*n + 1] * sin(ph);
            ci += signal[best_pos + 2*n]     * sin(ph) + signal[best_pos + 2*n + 1] * cos(ph);
        }
        corr_out[nid2] = sqrt(cr*cr + ci*ci);
    }

    *pss_pos = best_pos;
    return best_nid2;
}

void fft(double* re, double* im, int n) {
    if (n <= 1) return;
    int half = n / 2;
    double re_even[64], im_even[64], re_odd[64], im_odd[64];
    for (int i = 0; i < half; i++) {
        re_even[i] = re[2*i];   im_even[i] = im[2*i];
        re_odd[i]  = re[2*i+1]; im_odd[i]  = im[2*i+1];
    }
    fft(re_even, im_even, half);
    fft(re_odd,  im_odd,  half);
    for (int k = 0; k < half; k++) {
        double a = -2 * M_PI * k / n;
        double wr = cos(a), wi = sin(a);
        double tr = wr * re_odd[k] - wi * im_odd[k];
        double ti = wr * im_odd[k] + wi * re_odd[k];
        re[k]        = re_even[k] + tr;  im[k]        = im_even[k] + ti;
        re[k + half] = re_even[k] - tr;  im[k + half] = im_even[k] - ti;
    }
}

void gen_m_sequence(int* s) {
    int x[31] = {0, 0, 0, 0, 1};
    for (int j = 0; j < 26; j++) x[j + 5] = (x[j + 2] + x[j]) % 2;
    for (int i = 0; i < 31; i++) s[i] = 1 - 2 * x[i];
}

void cyclic_shift(int* s, int m, int* out) {
    for (int n = 0; n < 31; n++) out[n] = s[(n + m) % 31];
}

void gen_scrambling_c(int nid2, int* c0, int* c1) {
    int x[31] = {0, 0, 0, 0, 1};
    for (int j = 0; j < 26; j++) x[j + 5] = (x[j + 3] + x[j]) % 2;
    for (int i = 0; i < 31; i++) {
        c0[i] = 1 - 2 * x[(i + nid2) % 31];
        c1[i] = 1 - 2 * x[(i + nid2 + 3) % 31];
    }
}

void gen_scrambling_z(int m0, int* z1) {
    int x[31] = {0, 0, 0, 0, 1};
    for (int j = 0; j < 26; j++) x[j + 5] = (x[j + 4] + x[j + 2] + x[j + 1] + x[j]) % 2;
    for (int i = 0; i < 31; i++) z1[i] = 1 - 2 * x[(i + (m0 % 8)) % 31];
}

void gen_sss(int nid1, int nid2, int* d) {
    int s[31], s0[31], s1[31], c0[31], c1[31], z1[31];
    gen_m_sequence(s);
    int m0 = 15 * (nid1 / 112) + 5 * (nid1 % 8);
    int m1 = (nid1 % 112) + 5 * (nid1 / 112);
    cyclic_shift(s, m0, s0);
    cyclic_shift(s, m1, s1);
    gen_scrambling_c(nid2, c0, c1);
    gen_scrambling_z(m0, z1);
    for (int n = 0; n < 31; n++) {
        d[2*n]     = s0[n] * c0[n];
        d[2*n + 1] = s1[n] * c1[n] * z1[n];
    }
}

int detect_sss(const double* signal, int sss_pos, int nid2,
               int* best_nid1, double* best_corr) {
    double re[N_FFT], im[N_FFT];
    for (int i = 0; i < N_FFT; i++) {
        re[i] = signal[sss_pos + 2*i];
        im[i] = signal[sss_pos + 2*i + 1];
    }
    fft(re, im, N_FFT);

    int start = N_FFT / 2 - N_SYM / 2;
    double max_corr = 0;
    int best = 0;
    int d[62];

    for (int nid1 = 0; nid1 < 168; nid1++) {
        gen_sss(nid1, nid2, d);
        double cr = 0, ci = 0;
        for (int k = 0; k < N_SYM; k++) {
            cr += re[start + k] * d[k];
            ci += im[start + k] * d[k];
        }
        double c = sqrt(cr*cr + ci*ci);
        if (c > max_corr) {
            max_corr = c;
            best = nid1;
        }
    }
    *best_nid1 = best;
    *best_corr = max_corr;
    return 3 * best + nid2;
}

double* read_iq_file(const char* filename, int* samples_read) {
    FILE* f = fopen(filename, "rb");
    if (!f) return NULL;
    short* buf = (short*)malloc(N_SAMPLES * 2 * sizeof(short));
    size_t read = fread(buf, sizeof(short), N_SAMPLES * 2, f);
    fclose(f);
    *samples_read = (int)read / 2;

    double* signal = (double*)malloc(N_SAMPLES * 2 * sizeof(double));
    for (int i = 0; i < N_SAMPLES * 2 && i < read; i++)
        signal[i] = buf[i] / 32768.0;
    free(buf);
    return signal;
}

int main() {
    printf("=== LTE PCI Detector ===\n\n");

    int samples_read = 0;
    double* signal = read_iq_file(FILE_NAME, &samples_read);
    if (!signal) {
        printf("ERROR: cannot open '%s'\n", FILE_NAME);
        return 1;
    }
    printf("Loaded %d complex samples\n\n", samples_read);

    double pss[3][N_SYM];
    gen_pss(pss);

    int pss_pos = 0;
    double pss_corr[3];
    int nid2 = detect_pss(signal, N_SAMPLES, pss, &pss_pos, pss_corr);

    printf("PSS found at index %d\n", pss_pos / 2);
    printf("  Correlations:\n");
    for (int i = 0; i < 3; i++)
        printf("    N_ID(2)=%d (u=%d): %.3f\n", i, (int[]){25,29,34}[i], pss_corr[i]);
    printf("  Selected N_ID(2) = %d\n\n", nid2);

    int best_offset = 0;
    int best_pci = 0;
    int best_nid1 = 0;
    double best_corr = 0;

    printf("Searching SSS offset:\n");
    for (int offset = 120; offset <= 160; offset++) {
        int sss_pos = pss_pos - offset;
        if (sss_pos < 0) continue;

        int n1 = 0;
        double c = 0;
        int pci = detect_sss(signal, sss_pos, nid2, &n1, &c);

        printf("  offset=%3d  N_ID(1)=%3d  corr=%.1f  PCI=%d\n",
               offset, n1, c, pci);

        if (c > best_corr) {
            best_corr = c;
            best_nid1 = n1;
            best_offset = offset;
            best_pci = pci;
        }
    }

    printf("\nBest offset: %d\n", best_offset);
    printf("Best correlation: %.1f\n", best_corr);
    printf("N_ID(1) = %d\n", best_nid1);
    printf("N_ID(2) = %d\n", nid2);
    printf("PCI = 3 * %d + %d = %d\n\n", best_nid1, nid2, best_pci);

    printf("Running %d iterations...\n", RUNS);
    clock_t start_time = clock();

    int results[RUNS];
    double times[RUNS];

    for (int run = 0; run < RUNS; run++) {
        clock_t run_start = clock();

        int p_pos = 0;
        double p_c[3];
        int n2 = detect_pss(signal, N_SAMPLES, pss, &p_pos, p_c);

        int s_pos = p_pos - best_offset;
        if (s_pos < 0) s_pos = 0;

        int n1 = 0;
        double c = 0;
        results[run] = detect_sss(signal, s_pos, n2, &n1, &c);

        clock_t run_end = clock();
        times[run] = (double)(run_end - run_start) / CLOCKS_PER_SEC * 1000;
    }

    clock_t end_time = clock();
    double total_time = (double)(end_time - start_time) / CLOCKS_PER_SEC * 1000;

    printf("\nResults\n");
    int pci = results[0];
    printf("PCI: %d\n", pci);
    printf("  N_ID(2) = %d\n", pci % 3);
    printf("  N_ID(1) = %d\n", pci / 3);

    double min_time = times[0], max_time = times[0], sum_time = 0;
    for (int i = 0; i < RUNS; i++) {
        sum_time += times[i];
        if (times[i] < min_time) min_time = times[i];
        if (times[i] > max_time) max_time = times[i];
    }

    printf("\nPerformance (%d runs):\n", RUNS);
    printf("  Average: %.3f ms\n", sum_time / RUNS);
    printf("  Min:     %.3f ms\n", min_time);
    printf("  Max:     %.3f ms\n", max_time);
    printf("  Total:   %.3f ms\n", total_time);

    free(signal);
    return 0;
}
